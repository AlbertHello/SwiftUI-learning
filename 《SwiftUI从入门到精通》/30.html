SwiftData 模型代码示例

import SwiftData
import SwiftUI

@Model
final class TaskItem {
    var id: UUID
    var title: String
    var notes: String
    var isCompleted: Bool
    var priority: Int // 0:低, 1:中, 2:高
    var dueDate: Date?
    var createDate: Date
    @Relationship(deleteRule: .nullify) var category: Category?

    init(title: String, notes: String = "", priority: Int = 1) {
        self.id = UUID()
        self.title = title
        self.notes = notes
        self.isCompleted = false
        self.priority = priority
        self.createDate = Date()
    }
}

@Model
final class Category {
    var id: UUID
    var name: String
    var color: String // 存储十六进制颜色
    var icon: String // SF Symbols 名称

    init(name: String, color: String = "#4CAF50", icon: String = "folder") {
        self.id = UUID()
        self.name = name
        self.color = color
        self.icon = icon
    }
}

1. 任务列表行视图 (TaskRow)

struct TaskRow: View {
    let task: TaskItem
    @Binding var isEditing: Bool
    var onToggleComplete: () -> Void

    var body: some View {
        HStack(spacing: 15) {
            // 完成状态按钮
            Button(action: onToggleComplete) {
                Image(systemName: task.isCompleted ? "checkmark.circle.fill" : "circle")
                    .foregroundColor(task.isCompleted ? .green : .gray)
                    .font(.title2)
                    .animation(.easeInOut, value: task.isCompleted)
            }
            .buttonStyle(.plain)

            VStack(alignment: .leading, spacing: 5) {
                Text(task.title)
                    .font(.headline)
                    .strikethrough(task.isCompleted, color: .gray)
                    .foregroundColor(task.isCompleted ? .gray : .primary)

                if let dueDate = task.dueDate {
                    HStack {
                        Image(systemName: "calendar")
                            .font(.caption)
                        Text(dueDate, style: .date)
                            .font(.caption)
                            .foregroundColor(dueDate < Date() ? .red : .secondary)
                    }
                }
            }

            Spacer()

            // 优先级标签
            PriorityBadge(priority: task.priority)

            // 分类标签
            if let category = task.category {
                CategoryChip(category: category)
            }
        }
        .padding(.vertical, 10)
        .padding(.horizontal)
        .background(Color(.systemBackground))
        .cornerRadius(12)
        .shadow(color: .black.opacity(0.05), radius: 3, x: 0, y: 2)
    }
}

2. 视图模型示例 (TaskViewModel)

import SwiftData
import SwiftUI

@Observable
class TaskViewModel {
    private var modelContext: ModelContext
    var tasks: [TaskItem] = []
    var filter: FilterType = .all
    var searchText: String = ""

    enum FilterType {
        case all, pending, completed, highPriority
    }

    init(modelContext: ModelContext) {
        self.modelContext = modelContext
        fetchTasks()
    }

    func fetchTasks() {
        let descriptor = FetchDescriptor<TaskItem>(
            sortBy: [SortDescriptor(\.dueDate, order: .forward)]
        )
        do {
            tasks = try modelContext.fetch(descriptor)
        } catch {
            print("Fetch failed: \(error)")
        }
    }

    func addTask(title: String, priority: Int = 1) {
        let newTask = TaskItem(title: title, priority: priority)
        modelContext.insert(newTask)
        saveContext()
        fetchTasks()
    }

    func toggleComplete(for task: TaskItem) {
        task.isCompleted.toggle()
        saveContext()
    }

    func deleteTask(at offsets: IndexSet) {
        for index in offsets {
            let task = tasks[index]
            modelContext.delete(task)
        }
        saveContext()
        fetchTasks()
    }

    private func saveContext() {
        do {
            try modelContext.save()
        } catch {
            print("Save failed: \(error)")
        }
    }

    var filteredTasks: [TaskItem] {
        var result = tasks
        if !searchText.isEmpty {
            result = result.filter { $0.title.localizedCaseInsensitiveContains(searchText) }
        }
        switch filter {
        case .pending:
            return result.filter { !$0.isCompleted }
        case .completed:
            return result.filter { $0.isCompleted }
        case .highPriority:
            return result.filter { $0.priority == 2 }
        case .all:
            return result
        }
    }
}
